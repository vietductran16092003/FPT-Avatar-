# Generic Text Overlays Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the fixed 4-field `Template.components` (`joinYear`/`unit`/`slogan`/`signature`) with a generic, admin-authored `textOverlays` array, so a new Campaign/Template for a different event can declare whatever text fields it needs without touching code.

**Architecture:** `Template.components: Json` (string array) becomes `Template.textOverlays: Json` (array of self-describing objects: key, label, type, options, position, style). The server-side compositor gains SVG-based text rendering so the mandatory server re-composite (Task 8 of the prior plan) draws the real overlay text, not just the photo+frame. Admin gets a click-to-place UI on the frame preview to set each overlay's position instead of typing raw numbers. Public site renders its Step-3 form dynamically from whatever overlays the chosen Template declares.

**Tech Stack:** Same as the existing backend (Express 5 + TypeScript/NodeNext, Prisma + PostgreSQL, `sharp`) and existing FE (vanilla JS, no build step). No new dependencies.

**Spec:** [docs/superpowers/specs/2026-08-20-generic-text-overlays-design.md](../specs/2026-08-20-generic-text-overlays-design.md)

## Global Constraints

- Overlay values are user-supplied, untrusted input — every value composited into SVG text server-side must be XML-escaped before insertion (spec §3).
- Server always re-fetches `template.textOverlays` from the DB by the campaign-scoped template lookup already in place (`Backend/src/routes/generate.ts`) — never trusts a client-supplied overlay list (spec §3, §6).
- For an overlay with `type: "select"`, the submitted value must be a member of that overlay's own `options[]`, not merely a known key (spec §3).
- No data migration needed — no production data exists; schema changes replace `components`/`componentValues` outright and the seed script is rewritten to the new shape (spec §1, §6).
- Scope is text overlays only — no image/icon overlays, no drag-to-adjust (click-to-set only), no cross-template overlay duplication (spec §7).

- [ ] **Step 3: Implement text-overlay rendering**

Replace `Backend/src/compositor.ts` in full:

```ts
import sharp from "sharp";

export interface PhotoArea {
  x: number;
  y: number;
  w: number;
  h: number;
}

export interface TextOverlay {
  key: string;
  label: string;
  labelEn: string;
  type: "select" | "text";
  options?: string[];
  placeholder?: string;
  x: number;
  y: number;
  fontSize: number;
  color: string;
}

function escapeXml(value: string): string {
  return value
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;");
}

function buildOverlaySvg(
  overlays: TextOverlay[],
  values: Record<string, string>,
  width: number,
  height: number
): Buffer {
  const texts = overlays
    .filter((o) => values[o.key])
    .map((o) => {
      const px = (o.x / 100) * width;
      const py = (o.y / 100) * height;
      const safeValue = escapeXml(String(values[o.key]));
      return `<text x="${px}" y="${py}" font-size="${o.fontSize}" fill="${o.color}" text-anchor="middle" dominant-baseline="middle">${safeValue}</text>`;
    })
    .join("");
  return Buffer.from(`<svg width="${width}" height="${height}" xmlns="http://www.w3.org/2000/svg">${texts}</svg>`);
}

export async function compositeAvatar(
  photoBuffer: Buffer,
  framePngBuffer: Buffer,
  photoArea: PhotoArea,
  textOverlays: TextOverlay[],
  overlayValues: Record<string, string>
): Promise<Buffer> {
  const frameMeta = await sharp(framePngBuffer).metadata();
  const frameWidth = frameMeta.width ?? 500;
  const frameHeight = frameMeta.height ?? 500;

  const photoWidth = Math.round((photoArea.w / 100) * frameWidth);
  const photoHeight = Math.round((photoArea.h / 100) * frameHeight);
  const left = Math.round((photoArea.x / 100) * frameWidth);
  const top = Math.round((photoArea.y / 100) * frameHeight);

  const resizedPhoto = await sharp(photoBuffer)
    .resize(photoWidth, photoHeight, { fit: "cover" })
    .png()
    .toBuffer();

  const overlaySvg = buildOverlaySvg(textOverlays, overlayValues, frameWidth, frameHeight);

  return sharp({
    create: { width: frameWidth, height: frameHeight, channels: 4, background: { r: 0, g: 0, b: 0, alpha: 0 } },
  })
    .composite([
      { input: resizedPhoto, left, top },
      { input: overlaySvg, left: 0, top: 0 },
      { input: framePngBuffer, left: 0, top: 0 },
    ])
    .png()
    .toBuffer();
}
```

import { getStorage } from "./storage/index.js";
import type { Template } from "@prisma/client";

export interface PhotoArea {
  x: number;
  y: number;
  w: number;
  h: number;

export async function createTemplate(input: {
  campaignId: string;
  name: string;
  imageBuffer: Buffer;
  photoArea: PhotoArea;
  textOverlays: TextOverlay[];
}): Promise<Template> {
  const key = `frames/${input.campaignId}/${Date.now()}-${input.name.replace(/\s+/g, "-")}.png`;
  await getStorage().upload(key, input.imageBuffer, "image/png");
  return prisma.template.create({

      name: input.name,
      frameImageKey: key,
      photoArea: input.photoArea as unknown as object,
      textOverlays: input.textOverlays as unknown as object,
    },
  });
}

export async function updateTemplate(id: string, patch: Partial<{
  name: string; photoArea: PhotoArea; textOverlays: TextOverlay[];
}>): Promise<Template> {
  return prisma.template.update({
    where: { id },
    data: patch as unknown as Record<string, unknown>,
  });
}

export async function deleteTemplate(id: string): Promise<void> {
  const template = await prisma.template.findUnique({ where: { id } });
  if (!template) return;
  await prisma.template.delete({ where: { id } });
  await getStorage().delete(template.frameImageKey);
}

export async function withResolvedFrameUrl<T extends { frameImageKey: string }>(
  template: T
): Promise<T & { frameImageUrl: string }> {
  const frameImageUrl = await getStorage().getPublicUrl(template.frameImageKey);
  return { ...template, frameImageUrl };

```ts
import { Router } from "express";
import multer from "multer";
import { prisma } from "../db.js";
import { findCampaignBySlug } from "../campaigns.js";
import { getStorage } from "../storage/index.js";
import { compositeAvatar } from "../compositor.js";
import type { TextOverlay } from "../compositor.js";

const upload = multer({ storage: multer.memoryStorage() });

export const generateRouter = Router();

generateRouter.post("/:slug/generate", upload.single("photo"), async (req, res) => {
  const campaign = await findCampaignBySlug(String(req.params.slug));
  if (!campaign) return res.status(404).json({ error: "Campaign not found" });

  const { templateId } = req.body as { templateId: string };
  // Look up the template ONLY within this campaign's own templates array —
  // never trust a client-supplied templateId that could reference another
  // campaign's template (and thus another campaign's frame/photoArea).
  const template = campaign.templates.find((t) => t.id === (templateId ?? ""));
  if (!template) return res.status(400).json({ error: "Template does not belong to this campaign" });

  if (!req.file) return res.status(400).json({ error: "photo file is required" });

  let overlayValues: Record<string, unknown>;
  try {
    overlayValues = JSON.parse(req.body.overlayValues ?? "{}");
  } catch {
    return res.status(400).json({ error: "overlayValues must be valid JSON" });
  }
  if (typeof overlayValues !== "object" || overlayValues === null || Array.isArray(overlayValues)) {
    return res.status(400).json({ error: "overlayValues must be a JSON object" });
  }

  // Validate overlayValues against the template's OWN declared textOverlays —
  // this is the only client input allowed to influence the record, and only
  // its keys and (for select-type overlays) its values against that
  // overlay's own options — never the frame image, never the layout.
  const textOverlays = template.textOverlays as unknown as TextOverlay[];
  const overlayByKey = new Map(textOverlays.map((o) => [o.key, o]));

  for (const [key, value] of Object.entries(overlayValues)) {
    const overlay = overlayByKey.get(key);
    if (!overlay) {
      return res.status(400).json({ error: `Unknown overlay key: ${key}` });
    }
    if (overlay.type === "select" && !overlay.options?.includes(String(value))) {
      return res.status(400).json({ error: `Invalid value for overlay ${key}` });
    }
  }

  const storage = getStorage();

  // Fetch the REAL frame image bytes from storage using the key stored in
  // the DB for this template. The client cannot supply frame bytes or a
  // frame key — any such fields in the request body are ignored.
  const frameBuffer = await storage.download(template.frameImageKey);

  // Use the REAL photoArea from the DB — any client-supplied photoArea in
  // the request body is ignored.
  const photoArea = template.photoArea as unknown as { x: number; y: number; w: number; h: number };
  const resultBuffer = await compositeAvatar(
    req.file.buffer,
    frameBuffer,
    photoArea,
    textOverlays,
    overlayValues as Record<string, string>
  );

  const resultKey = `results/${campaign.id}/${Date.now()}.png`;
  await storage.upload(resultKey, resultBuffer, "image/png");
  const resultUrl = await storage.getPublicUrl(resultKey);

  await prisma.generatedAvatar.create({
    data: {
      campaignId: campaign.id,
      templateId: template.id,
      overlayValues: overlayValues as unknown as object,
      resultImageKey: resultKey,
    },
  });
  await prisma.campaign.update({ where: { id: campaign.id }, data: { downloadCount: { increment: 1 } } });

  res.status(201).json({ resultUrl });
});
```
